# --------------------------------------------------------------
# PARTE 1: CLIENTE DESKTOP LOCAL (Seu Computador)

# 1. Iniciar o Daemon Local (Rust + Axum)
cd /home/vanir/fury/frontend/backend
cargo run
# Endpoint: http://127.0.0.1:12540
# WebSocket: ws://127.0.0.1:12540/api/ws

# 2. Iniciar a Interface Desktop (Vite + React)
cd /home/vanir/fury/frontend/app
npm run dev
# URL: http://localhost:1254

# --------------------------------------------------------------
# PARTE 2: SERVIÇOS EM NUVEM / VPS (/vps)
# --------------------------------------------------------------

# 3. Subir o Armazenamento de Objetos S3 (MinIO)
cd /home/vanir/fury/frontend/vps/server-backend
docker compose up -d minio
# MinIO S3 API: http://localhost:8987
# MinIO Console Web: http://localhost:8988 (User: furyadmin / Pass: furypassword123)

# 4. Iniciar a API Gateway Cloud Sync (NestJS + Prisma SQLite)
cd /home/vanir/fury/frontend/vps/server-backend
npm run start:dev
# URL: http://0.0.0.0:12342

# 5. Iniciar o Portal Web de Autenticação e Registro (Next.js)
cd /home/vanir/fury/frontend/vps/site
npm run dev
# URL: http://localhost:3001
